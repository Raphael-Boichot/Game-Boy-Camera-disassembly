_RET
	LD	A,$10
	LD	($4000),A	;RAM BANK 10H
	XOR	A
	RET


;=================
;OFFSET_DOWN
;=================
OFFSET_DOWN
	LD	A,(B_CNTR5)

	BIT	5,A		;符号チェック
	JR	Z,_MINUS

;------------------------------
_PLUS
	AND	$3F		;下位6BITをみて
	CP	$20
	JR	Z,_PLUS_END	;現在＋０の時は・・・_PLUS_END
	
	DEC	A		;減らす
;	SET	5,A		;符号ビットをたてる
	LD	B,$00
	JR	_DOWN_END
	
_PLUS_END			;-1にする
	LD	A,$01
	LD	B,$00
	JR	_DOWN_END

;------------------------------
_MINUS
	AND	$1F
	CP	$1F
	JR	NZ,_MINUS_DOWN
	LD	B,$01		;終了フラグ
	JR	_DOWN_END

_MINUS_DOWN
	INC	A		;へらす
	LD	B,$00

;-----------------------------
_DOWN_END
	LD	C,A
	LD	A,(B_CNTR5)
	AND	$C0
	OR	C
	LD	(B_CNTR5),A
	LD	(CNTR5),A

	LD	A,B		;一番下までいったかフラグ

	
	RET
